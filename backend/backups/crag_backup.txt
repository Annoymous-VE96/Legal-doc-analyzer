from __future__ import annotations

import os
import re
from typing import List, TypedDict, Annotated, Optional
from operator import add

from dotenv import load_dotenv
from pydantic import BaseModel, field_validator

from langchain_core.documents import Document
from langchain_core.prompts import ChatPromptTemplate
from langchain_community.document_loaders import PyPDFLoader
from langchain_community.vectorstores import FAISS
from langchain_huggingface import HuggingFaceEmbeddings
from langchain_groq import ChatGroq
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.tools.tavily_search import TavilySearchResults
from langgraph.graph import StateGraph, START, END

from pathlib import Path
load_dotenv(Path(__file__).parent.parent / ".env", override=True)

class State(TypedDict, total=False):
    question: str
    docs: List[Document]
    good_docs: List[Document]
    verdict: str
    reason: str
    strips: List[str]
    kept_strips: List[str]
    refined_context: str
    web_docs: Annotated[List[Document], add]
    web_query: str
    answer: str


class ChunkScore(BaseModel):
    chunk_index: int
    score: float
    reason: str

class BatchDocsEvalScore(BaseModel):
    scores: List[ChunkScore]


class SentenceVerdict(BaseModel):
    sentence_index: int
    keep: bool

class BatchKeepOrDrop(BaseModel):
    verdicts: List[SentenceVerdict]


class WebQuery(BaseModel):
    query: str


class CRAGPipeline:
    def __init__(
        self,
        pdf_path: str,
        filename: str,
        upload_dir: str = "../uploads",
        vector_store_dir: str = "../vector_store",
        chunk_size: int = 900,
        chunk_overlap: int = 100,
        upper_th: float = 0.7,
        lower_th: float = 0.3,
        llm_model: str = "llama-3.3-70b-versatile",
        embedding_model: str = "all-MiniLM-L6-v2",
        temperature: float = 0.5,
    ):
        load_dotenv("../.env", override=True)

        self.pdf_path = pdf_path
        self.filename = filename
        self.upload_dir = upload_dir
        self.vector_store_dir = vector_store_dir
        self.upper_th = upper_th
        self.lower_th = lower_th

        self.loader = PyPDFLoader(self.pdf_path)
        self.splitter = RecursiveCharacterTextSplitter(
            chunk_size=chunk_size,
            chunk_overlap=chunk_overlap,
        )

        self.embeddings = HuggingFaceEmbeddings(model=embedding_model)
        self.llm = ChatGroq(model=llm_model, temperature=temperature)
        self.tavily = TavilySearchResults(max_results=5)

        self.docs: List[Document] = []
        self.chunks: List[Document] = []
        self.vector_store: Optional[FAISS] = None
        self.retriever = None

        self.doc_eval_chain = self._build_doc_eval_chain()
        self.filter_chain = self._build_filter_chain()
        self.rewrite_chain = self._build_rewrite_chain()
        self.answer_chain = self._build_answer_chain()

        self._prepare_knowledge_base()
        self.app = self._build_graph()

    # -----------------------
    # 1) Loader
    # -----------------------
    def load_documents(self) -> List[Document]:
        self.docs = self.loader.load()
        return self.docs

    # -----------------------
    # 2) Chunker
    # -----------------------
    def chunk_documents(self) -> List[Document]:
        if not self.docs:
            self.load_documents()

        self.chunks = self.splitter.split_documents(self.docs)

        for chunk in self.chunks:
            chunk.metadata["source"] = self.filename

        return self.chunks

    # -----------------------
    # 3) Embedder + Vector Store
    # -----------------------
    def build_vector_store(self) -> FAISS:
        if not self.chunks:
            self.chunk_documents()

        self.vector_store = FAISS.from_documents(self.chunks, self.embeddings)

        os.makedirs(self.vector_store_dir, exist_ok=True)
        self.vector_store.save_local(self.vector_store_dir)

        self.vector_store = FAISS.load_local(
            self.vector_store_dir,
            self.embeddings,
            allow_dangerous_deserialization=True,
        )

        self.retriever = self.vector_store.as_retriever(
            search_type="similarity",
            search_kwargs={"k": 4},
        )
        return self.vector_store

    def _prepare_knowledge_base(self) -> None:
        if os.path.exists(self.vector_store_dir):
            # load from disk
            self.vector_store = FAISS.load_local(
                self.vector_store_dir,
                self.embeddings,
                allow_dangerous_deserialization=True
            )
        else:
            # first time — embed and save
            self.load_documents()
            self.chunk_documents()
            self.build_vector_store()

        # set retriever in both cases
        self.retriever = self.vector_store.as_retriever(
            search_type="similarity",
            search_kwargs={"k": 4}
        )


    # -----------------------
    # LLM Chains
    # -----------------------
    def _build_doc_eval_chain(self):
        prompt = ChatPromptTemplate.from_messages(
            [
                (
                    "system",
                    "You are a strict retrieval evaluator for CRAG.\n"
                    "You will be given MULTIPLE retrieved chunks and a question.\n"
                    "For EACH chunk, return its chunk_index (0-based), a relevance score in [0,1], and a short reason.\n"
                    "1 = mostly relevant, 0 = irrelevant.\n"
                    "You MUST return a score for every chunk provided.",
                ),
                ("human", "Question: {question}\n\nChunks:\n{chunks}"),
            ]
        )
        return prompt | self.llm.with_structured_output(BatchDocsEvalScore)

    def _build_filter_chain(self):
        prompt = ChatPromptTemplate.from_messages(
            [
                (
                    "system",
                    "You are a strict relevance filter.\n"
                    "You will be given MULTIPLE numbered sentences and a question.\n"
                    "For EACH sentence, return its sentence_index (0-based) and keep=true only if it directly helps answer the question.\n"
                    "You MUST return a verdict for every sentence provided.\n"
                    "Output JSON only.",
                ),
                ("human", "Question: {question}\n\nSentences:\n{sentences}"),
            ]
        )
        return prompt | self.llm.with_structured_output(BatchKeepOrDrop)

    def _build_rewrite_chain(self):
        prompt = ChatPromptTemplate.from_messages(
            [
                (
                    "system",
                    "You are a search query optimizer.\n"
                    "Rewrite the given question into a concise, search-engine-friendly query.\n"
                    "Output JSON only.",
                ),
                ("human", "Question: {question}"),
            ]
        )
        return prompt | self.llm.with_structured_output(WebQuery)

    def _build_answer_chain(self):
        prompt = ChatPromptTemplate.from_messages(
            [
                (
                    "system",
                    "You are a helpful Advocate/Lawyer. Answer only using the provided context.\n"
                    "If the context is empty or insufficient, say: I don't know",
                ),
                ("human", "Question: {question}\n\nRefined context:\n{refined_context}"),
            ]
        )
        return prompt | self.llm

    # -----------------------
    # Retriever node
    # -----------------------
    def retrieve(self, state: State) -> State:
        q = state["question"]
        return {"docs": self.retriever.invoke(q)}

    # -----------------------
    # Document evaluation
    # -----------------------
    def eval_each_docs(self, state: State) -> State:
        q = state["question"]
        docs = state.get("docs", [])

        # Build a single prompt with all chunks labeled by index
        chunks_text = "\n\n".join(
            f"[Chunk {i}]:\n{d.page_content}"
            for i, d in enumerate(docs)
        )

        # 1 LLM call for all chunks
        output = self.doc_eval_chain.invoke({"question": q, "chunks": chunks_text})

        # Map index -> score from batch response
        score_map = {cs.chunk_index: cs.score for cs in output.scores}

        good_docs: List[Document] = []
        scores: List[float] = []

        for i, d in enumerate(docs):
            score = score_map.get(i, 0.0)
            scores.append(score)
            if score > self.lower_th:
                good_docs.append(d)

        if any(s > self.upper_th for s in scores):
            return {
                "good_docs": good_docs,
                "verdict": "CORRECT",
                "reason": f"At least one document scored above {self.upper_th}",
            }

        if scores and all(s < self.lower_th for s in scores):
            return {
                "good_docs": [],
                "verdict": "INCORRECT",
                "reason": f"None of the docs scored above {self.lower_th}",
            }

        return {
            "good_docs": good_docs,
            "verdict": "AMBIGUOUS",
            "reason": f"All docs are between {self.upper_th} and {self.lower_th}",
        }

    # -----------------------
    # Sentence splitting
    # -----------------------
    def tokenize_sentences(self, text: str) -> List[str]:
        text = re.sub(r"\s+", " ", text).strip()
        sentences = re.split(r"(?<=[.!?])\s+", text)
        return [s.strip() for s in sentences if len(s.strip()) > 20]

    # -----------------------
    # Knowledge refinement
    # -----------------------
    def refine(self, state: State) -> State:
        q = state["question"]
        verdict = state.get("verdict", "")

        if verdict == "CORRECT":
            docs_to_use = state.get("good_docs", [])
        elif verdict == "INCORRECT":
            docs_to_use = state.get("web_docs", [])
        else:
            docs_to_use = state.get("good_docs", []) + state.get("web_docs", [])

        context = "\n\n".join(d.page_content for d in docs_to_use).strip()
        strips = self.tokenize_sentences(context)

        # Build a single prompt with all sentences labeled by index
        sentences_text = "\n".join(
            f"[Sentence {i}]: {s}"
            for i, s in enumerate(strips)
        )

        # 1 LLM call for all sentences
        output = self.filter_chain.invoke({"question": q, "sentences": sentences_text})

        # Map index -> keep from batch response
        keep_map = {sv.sentence_index: sv.keep for sv in output.verdicts}

        kept: List[str] = [
            s for i, s in enumerate(strips)
            if keep_map.get(i, False)
        ]

        refined_context = "\n".join(kept).strip()

        return {
            "strips": strips,
            "kept_strips": kept,
            "refined_context": refined_context,
        }

    # -----------------------
    # Query rewrite
    # -----------------------
    def rewrite_query_node(self, state: State) -> State:
        output = self.rewrite_chain.invoke({"question": state["question"]})
        return {"web_query": output.query}

    # -----------------------
    # Web search
    # -----------------------
    def web_search_node(self, state: State) -> State:
        q = state.get("web_query") or state["question"]
        results = self.tavily.invoke({"query": q})

        web_docs: List[Document] = []
        for r in results or []:
            title = r.get("title", "")
            url = r.get("url", "")
            content = r.get("content", "") or r.get("snippet", "")

            text = f"TITLE: {title}\nURL: {url}\nCONTENT:\n{content}"
            web_docs.append(
                Document(
                    page_content=text,
                    metadata={"url": url, "title": title, "source": "web"},
                )
            )

        return {"web_docs": web_docs}

    # -----------------------
    # Answer generation
    # -----------------------
    def generate(self, state: State) -> State:
        output = self.answer_chain.invoke(
            {
                "question": state["question"],
                "refined_context": state.get("refined_context", ""),
            }
        )
        return {"answer": output.content}

    # -----------------------
    # Routing
    # -----------------------
    def route_after_eval(self, state: State) -> str:
        if state["verdict"] == "CORRECT":
            return "refine"
        return "rewrite_query"

    # -----------------------
    # Build graph
    # -----------------------
    def _build_graph(self):
        g = StateGraph(State)

        g.add_node("retrieve", self.retrieve)
        g.add_node("eval_each_doc", self.eval_each_docs)
        g.add_node("rewrite_query", self.rewrite_query_node)
        g.add_node("web_search", self.web_search_node)
        g.add_node("refine", self.refine)
        g.add_node("generate", self.generate)

        g.add_edge(START, "retrieve")
        g.add_edge("retrieve", "eval_each_doc")

        g.add_conditional_edges(
            "eval_each_doc",
            self.route_after_eval,
            {
                "refine": "refine",
                "rewrite_query": "rewrite_query",
            },
        )

        g.add_edge("rewrite_query", "web_search")
        g.add_edge("web_search", "refine")
        g.add_edge("refine", "generate")
        g.add_edge("generate", END)

        return g.compile()

    # -----------------------
    # Run pipeline
    # -----------------------
    def run(self, question: str) -> State:
        initial_state: State = {
            "question": question,
            "docs": [],
            "good_docs": [],
            "web_docs": [],
            "verdict": "",
            "reason": "",
            "strips": [],
            "kept_strips": [],
            "refined_context": "",
            "answer": "",
        }
        return self.app.invoke(initial_state)
